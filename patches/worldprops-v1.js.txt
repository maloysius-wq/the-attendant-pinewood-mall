function pickupFallback(type){
  const g=new THREE.Group();
  const metal=new THREE.MeshStandardMaterial({color:0xb8babd,roughness:.42,metalness:.42});
  const paper=new THREE.MeshStandardMaterial({color:0xd9d1bb,roughness:.82});
  const dark=new THREE.MeshStandardMaterial({color:0x25242a,roughness:.62,metalness:.12,emissive:type==='decoy'?0x4e1536:0x000000,emissiveIntensity:type==='decoy'?.45:0});
  if(type==='key'){
    const shaft=new THREE.Mesh(new THREE.CylinderGeometry(.018,.018,.22,10),metal);shaft.rotation.z=Math.PI/2;g.add(shaft);
    const ring=new THREE.Mesh(new THREE.TorusGeometry(.065,.014,8,18),metal);ring.position.x=-.13;g.add(ring);
  }else if(type==='journal'){
    addBox(g,new THREE.Vector3(0,.03,0),new THREE.Vector3(.28,.06,.21),new THREE.MeshStandardMaterial({color:0x49352d,roughness:.85}));
    addBox(g,new THREE.Vector3(.015,.065,0),new THREE.Vector3(.24,.018,.18),paper);
  }else if(type==='note'){
    addBox(g,new THREE.Vector3(0,.012,0),new THREE.Vector3(.25,.024,.18),paper);
  }else if(type==='tape'){
    addBox(g,new THREE.Vector3(0,.035,0),new THREE.Vector3(.27,.07,.17),dark);
    for(const sx of[-1,1]){const reel=new THREE.Mesh(new THREE.TorusGeometry(.034,.009,7,14),paper);reel.rotation.x=Math.PI/2;reel.position.set(.07*sx,.074,0);g.add(reel);}
  }else{
    addBox(g,new THREE.Vector3(0,.055,0),new THREE.Vector3(.25,.11,.16),dark);
    const antenna=new THREE.Mesh(new THREE.CylinderGeometry(.007,.007,.18,7),metal);antenna.position.set(.085,.18,0);antenna.rotation.z=-.22;g.add(antenna);
  }
  return g;
}
async function attachPickupModel(group,type){
  const spec=PICKUP_MODEL_SPECS[type];if(!spec)return;
  let model=await assets.model(spec.model);
  if(group.userData.collected)return;
  if(!model){group.add(pickupFallback(type));group.userData.assetFallback=true;return;}
  model.rotation.set(spec.rx||0,spec.ry||0,spec.rz||0);
  model.position.set(0,0,0);
  const lift=normalizeModelToMeters(model,{targetLongest:spec.longest});
  model.position.y=lift;
  model.updateWorldMatrix(true,true);
  const box=new THREE.Box3().setFromObject(model),center=box.getCenter(new THREE.Vector3());
  model.position.x-=center.x;model.position.z-=center.z;
  model.traverse(o=>{if(o.isMesh){o.castShadow=true;o.receiveShadow=true;}});
  group.add(model);group.userData.assetModel=spec.model;
}
