async function placeCassetteGrounded(parent,name,pos,{targetHeight,rot=0,collide=true,navBlock=true,pad=.045,tag=null}={}){

  m.rotation.set(0,rot,0);m.scale.setScalar(1);m.position.set(pos.x,0,pos.z);parent.add(m);

function cassetteShelfLevels(fixture,desiredRows=4){
  const overall=cassetteWorldBox(fixture),size=overall.getSize(new THREE.Vector3()),height=size.y,candidates=[];
  fixture.traverse(o=>{
    if(!o.isMesh)return;
    const b=cassetteWorldBox(o),s=b.getSize(new THREE.Vector3());
    if(s.y<=height*.14&&(s.x>=size.x*.42||s.z>=size.z*.42)){
      const y=b.max.y;if(y>overall.min.y+height*.10&&y<overall.max.y-height*.04)candidates.push(y);
    }
  });
  candidates.sort((a,b)=>a-b);
  const clustered=[];for(const y of candidates)if(!clustered.some(v=>Math.abs(v-y)<.035))clustered.push(y);
  const usable=clustered.filter(y=>y<overall.max.y-height*.08);
  if(usable.length>=desiredRows){
    const out=[];for(let i=0;i<desiredRows;i++){const idx=Math.round(i*(usable.length-1)/Math.max(1,desiredRows-1));out.push(usable[idx]);}return out;
  }
  const fallback=[.18,.39,.60,.81];return fallback.slice(0,desiredRows).map(f=>overall.min.y+height*f);
}
async function stockCassetteFixture(world,fixture,rot,{rows=4,cols=9,density=.82}={}){
  const template=await getCassetteTapeTemplate();
  if(!template){console.warn('Cassette Castle tape component unavailable; leaving the shelf empty rather than floating fake stock.');return null;}
  const box=cassetteWorldBox(fixture),size=box.getSize(new THREE.Vector3()),center=box.getCenter(new THREE.Vector3());
  const quarter=Math.abs(Math.sin(rot))>.5,longSpan=quarter?size.z:size.x,depthSpan=quarter?size.x:size.z;
  const longVec=new THREE.Vector3(Math.cos(rot),0,-Math.sin(rot)),frontVec=new THREE.Vector3(Math.sin(rot),0,Math.cos(rot));
  const levels=cassetteShelfLevels(fixture,rows),stock=new THREE.Group();stock.userData.assetModel='poly-haven-cassette-tape-stock-v13';world.root.add(stock);
  for(let row=0;row<levels.length;row++)for(let col=0;col<cols;col++){
    const gate=((row*cols+col)*37+11)%100;if(gate>density*100)continue;
    const along=cols===1?0:(-longSpan*.37+col*((longSpan*.74)/(cols-1)));
    const x=center.x+longVec.x*along+frontVec.x*depthSpan*.31,z=center.z+longVec.z*along+frontVec.z*depthSpan*.31;
    const wobble=((row*cols+col)%3-1)*.012;
    addAnchoredCassette(stock,template,x,z,levels[row]+.008,rot+wobble,.96+((row+col)%3)*.015);
  }
  return stock;
}
